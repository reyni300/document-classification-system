ALLOWED_EXTENSIONS = {'png', 'jpg', 'jpeg', 'pdf'}

def allowed_file(filename):
    return '.' in filename and filename.rsplit('.', 1)[1].lower() in ALLOWED_EXTENSIONS


def normalize_for_search(text):
    """Нормализует строку для нечёткого сравнения."""
    if not text:
        return ''
    text = str(text).lower().strip()
    text = re.sub(r'[^\w\d]', '', text)
    return text


def find_blocks_for_value(blocks, value):
    """
    Ищет OCR-блоки, соответствующие извлечённому значению.
    Возвращает список блоков с координатами.
    """
    if not value or not blocks:
        return []

    value_str = str(value).strip()
    norm_value = normalize_for_search(value_str)

    if not norm_value or len(norm_value) < 2:
        return []

    matched = []

    # 1. Точное совпадение нормализованного текста блока
    for block in blocks:
        norm_block = normalize_for_search(block.get('text', ''))
        if norm_block and norm_value in norm_block:
            matched.append(block)

    if matched:
        return matched

    # 2. Нечёткий поиск: разбиваем значение на части и ищем по словам
    # Например "7260.00" -> ищем "7260", "7.260", "7 260" и т.д.
    # Убираем незначимые нули и знаки
    value_parts = re.findall(r'\d+', norm_value)
    if value_parts:
        main_number = value_parts[0]  # главное число
        if len(main_number) >= 3:
            for block in blocks:
                norm_block = normalize_for_search(block.get('text', ''))
                if main_number in norm_block:
                    matched.append(block)

    if matched:
        return matched

    # 3. Поиск по первым значимым символам (для компаний, дат и т.д.)
    # Берём первые 4+ символа и ищем
    if len(norm_value) >= 4:
        prefix = norm_value[:4]
        for block in blocks:
            norm_block = normalize_for_search(block.get('text', ''))
            if norm_block and prefix in norm_block:
                matched.append(block)

    return matched


def build_field_mapping(parsed_item, blocks):
    """
    Для каждого поля из parsed_item находит соответствующие OCR-блоки.
    Возвращает dict: { field_name: [{ text, left, top, width, height, confidence }] }
    """
    fields_to_map = ['total', 'date', 'company', 'invoice_number', 'vat', 'currency']
    mapping = {}

    # Основные поля
    for field in fields_to_map:
        value = parsed_item.get(field)
        if value and str(value).strip() and str(value).strip() not in ('N/A', 'None', ''):
            found_blocks = find_blocks_for_value(blocks, value)
            if found_blocks:
                mapping[field] = found_blocks

    # extra_fields — ключ формата "extra::<название поля>"
    extra = parsed_item.get('extra_fields') or {}
    for label, value in extra.items():
        if value and str(value).strip() not in ('N/A', 'None', '', '-'):
            found_blocks = find_blocks_for_value(blocks, value)
            if found_blocks:
                mapping[f'extra::{label}'] = found_blocks

    return mapping

@app.route('/api/process', methods=['POST'])
def process_document():
    if 'file' not in request.files:
        return jsonify({'error': 'Файл не передан'}), 400

    file = request.files['file']
    if file.filename == '':
        return jsonify({'error': 'Имя файла пустое'}), 400
    if not allowed_file(file.filename):
        return jsonify({'error': f'Формат не поддерживается: {file.filename}'}), 400

    suffix = '.' + file.filename.rsplit('.', 1)[1].lower()
    with tempfile.NamedTemporaryFile(delete=False, suffix=suffix) as tmp:
        file.save(tmp.name)
        tmp_path = tmp.name

    try:
        # Сбрасываем кэш LLM перед каждым новым файлом
        llm_agent._last_ocr_hash = None
        llm_agent._last_result = None

        # Получаем OCR-блоки ДО оркестратора, пока файл точно на диске
        try:
            ocr_data_for_mapping = ocr.extract(tmp_path)
            pages_blocks = [page.get('blocks', []) for page in ocr_data_for_mapping.get('pages', [])]
        except Exception:
            pages_blocks = []

        # Сбрасываем кэш снова — чтобы оркестратор запустил LLM заново,
        # а не вернул кэш от вызова выше
        llm_agent._last_ocr_hash = None
        llm_agent._last_result = None

        # Запускаем основную обработку
        result = orchestrator.process_document(tmp_path, file.filename)

        # Строим маппинг для каждой записи в parsed_list
        parsed_list = result.get('parsed_list', [])
        field_mappings = []

        for idx, parsed_item in enumerate(parsed_list):
            # Определяем страницу для этой записи
            page_idx = parsed_item.get('_page', idx + 1) - 1
            blocks = pages_blocks[page_idx] if page_idx < len(pages_blocks) else \
                     (pages_blocks[0] if pages_blocks else [])

            mapping = build_field_mapping(parsed_item, blocks)
            field_mappings.append(mapping)

        # Добавляем блоки страниц и маппинг в результат
        result['pages_blocks'] = pages_blocks
        result['field_mappings'] = field_mappings

        return jsonify(result)

    except Exception as e:
        return jsonify({
            'error': str(e),
            'filename': file.filename,
            'parsed_list': [],
            'logs': [],
            'pages_blocks': [],
            'field_mappings': []
        }), 500
    finally:
        try:
            os.unlink(tmp_path)
        except:
            pass


@app.route('/api/history', methods=['GET'])
def get_history():
    docs = db.get_all()
    result = []
    for doc_id, filename, created_at in docs:
        row = db.get_one(doc_id)
        total, currency, category = '—', '', 'прочее'
        if row:
            _, parsed = row
            first = parsed[0] if isinstance(parsed, list) and parsed else parsed
            if isinstance(first, dict):
                total    = first.get('total', '—')
                currency = first.get('currency', '')
                category = first.get('category', 'прочее')
        result.append({
            'id':       doc_id,
            'filename': filename,
            'date':     (created_at or '')[:10],
            'total':    total,
            'currency': currency,
            'category': category,
        })
    return jsonify(result)

import json
import requests


class LLMClient:
    def __init__(self, model="mistral"):
        self.model = model
        self.url = "http://localhost:11434/api/generate"

    def generate(self, prompt: str):
        try:
            response = requests.post(
                self.url,
                json={
                    "model": self.model,
                    "prompt": prompt,
                    "stream": False,
                    "format": "json"
                },
                timeout=120
            )

            response.raise_for_status()
            data = response.json()

            if "response" not in data:
                print("ERROR RESPONSE:", data)
                return {"category": "прочее"}

            result = data["response"]
            print("\n[LLM RAW OUTPUT]:", result)

            return self.clean_json(result)

        except Exception as e:
            print("LLM ERROR:", e)
            return {"category": "прочее"}

    def clean_json(self, text: str):
        try:
            return json.loads(text)
        except Exception:
            start = text.find("{")
            end = text.rfind("}")

            if start != -1 and end != -1:
                try:
                    return json.loads(text[start:end + 1])
                except Exception:
                    pass

            return {"category": "прочее"}

import json

def clean_json_trailing_commas(json_str):
    json_str = re.sub(r',\s*}', '}', json_str)
    json_str = re.sub(r',\s*]', ']', json_str)
    return json_str


class InvoiceData(BaseModel):
    date: Optional[str] = None
    total: Optional[str] = None
    currency: str = "RUB"
    invoice_number: Optional[str] = None
    company: Optional[str] = None
    vat: Optional[str] = None
    document_type: Optional[str] = None      # тип документа (чек, квитанция, счёт и т.д.)
    extra_fields: Optional[Dict] = None      # любые дополнительные поля которые нашёл LLM

    @validator('total', pre=True)
    def normalize_total(cls, v):
        if v is None:
            return None
        if isinstance(v, (int, float)):
            return f"{v:.2f}"
        v_str = str(v).strip()
        v_str = v_str.replace(' ', '').replace(',', '.')
        v_str = re.sub(r'[^\d.-]', '', v_str)
        try:
            return f"{float(v_str):.2f}"
        except ValueError:
            return None

    @validator('date', pre=True)
    def normalize_date(cls, v):
        if v is None:
            return None
        v_str = str(v).strip()
        if ' ' in v_str:
            v_str = v_str.split()[0]
        v_str = v_str.replace(' г.', '').strip()
        return v_str

    @validator('currency', pre=True)
    def normalize_currency(cls, v):
        if v is None or v == '':
            return 'RUB'
        v_str = str(v).strip().upper()

        if v_str in ['RUB', 'USD', 'EUR', 'BYN']:
            return v_str
        if v_str in ['РУБ', 'Р', 'РУБЛЬ', 'РУБЛЕЙ', 'RUR']:
            return 'RUB'
        if v_str in ['$', 'ДОЛЛ', 'ДОЛЛАР', 'DOLLARS']:
            return 'USD'
        if v_str in ['€', 'ЕВРО', 'EURO']:
            return 'EUR'
        if v_str in [
            'BYN', 'BYR',
            'ВУМ', 'ВУН', 'BУМ', 'BУН',
            'VUM', 'VUN', 'BUM', 'BUN',
            'БЕЛ', 'БЕЛРУБ', 'Б.Р.', 'БР',
            'БЕЛОРУССКИЙ', 'BEL',
        ]:
            return 'BYN'
        return 'RUB'

    @validator('invoice_number', pre=True)
    def normalize_number(cls, v):
        if v is None:
            return ''
        v_str = str(v).strip()
        v_str = re.sub(r'[^\w\-]', '', v_str)
        return v_str

    @validator('company', pre=True)
    def normalize_company(cls, v):
        if v is None:
            return ''
        v_str = str(v).strip()
        v_str = v_str.replace('"', '').replace('\n', ' ').strip()
        v_str = re.sub(r'0+', 'О', v_str)
        v_str = v_str.replace('`', "'").replace('\u2018', "'").replace('\u2019', "'")
        v_str = v_str.strip("'")
        return v_str

    @validator('vat', pre=True)
    def normalize_vat(cls, v):
        if v is None or v == '':
            return ''
        v_str = str(v).strip()
        v_str = re.sub(r'НДС\s*', '', v_str, flags=re.IGNORECASE)
        v_str = v_str.replace(' ', '').replace(',', '.')
        v_str = re.sub(r'[^\d.-]', '', v_str)
        try:
            return f"{float(v_str):.2f}"
        except:
            return v_str

    @validator('document_type', pre=True)
    def normalize_document_type(cls, v):
        if v is None:
            return None
        return str(v).strip()

    @validator('extra_fields', pre=True)
    def normalize_extra_fields(cls, v):
        if v is None:
            return {}
        if not isinstance(v, dict):
            return {}
        # Чистим: убираем пустые значения, None, "N/A"
        cleaned = {}
        for key, val in v.items():
            if val is None:
                continue
            val_str = str(val).strip()
            if val_str in ('', 'N/A', 'null', 'None', '-'):
                continue
            # Ключ — читаемое название
            cleaned_key = str(key).strip().replace('_', ' ').capitalize()
            cleaned[cleaned_key] = val_str
        return cleaned


class LLMAgent:
    def __init__(self):
        self._last_ocr_hash = None
        self._last_result = None

    def _get_ocr_hash(self, ocr_json):
        import hashlib
        text = ''.join([page['text'][:100] for page in ocr_json['pages']])
        return hashlib.md5(text.encode()).hexdigest()

    def interpret_multi(self, ocr_json):
        pages_text = [page['text'] for page in ocr_json['pages']]
        num_pages = len(pages_text)

        current_hash = self._get_ocr_hash(ocr_json)
        if self._last_ocr_hash == current_hash:
            print(f"DEBUG: Возвращаем кэшированный результат для {num_pages} страниц")
            return self._last_result

        print(f"DEBUG interpret_multi: {num_pages} страниц")

        if num_pages == 0:
            return []

        if num_pages == 1:
            result = self._extract_from_text(pages_text[0], full_text=pages_text[0], page_num=1)
            final_result = [result] if result and "error" not in result else []
        else:
            all_results = []
            for i, page_text in enumerate(pages_text):
                has_invoice = re.search(r'СЧЕТ\s*НА\s*ОПЛАТУ|Счет\s*№|№\s*\d+', page_text, re.IGNORECASE)
                if has_invoice:
                    result = self._extract_from_text(page_text, full_text=page_text, page_num=i + 1)
                    if result and "error" not in result:
                        all_results.append(result)
                else:
                    if all_results and i > 0:
                        combined_text = pages_text[i - 1] + "\n\n" + page_text
                        result = self._extract_from_text(combined_text, full_text=combined_text, page_num=i)
                        if result and "error" not in result and len(all_results) > 0:
                            all_results[-1] = result

            if not all_results:
                combined_text = "\n\n".join(pages_text)
                result = self._extract_from_text(combined_text, full_text=combined_text, page_num=1)
                if result and "error" not in result:
                    all_results = [result]

            for i, result in enumerate(all_results):
                result['_page'] = i + 1
                result['_total_pages'] = len(all_results)

            final_result = all_results

        self._last_ocr_hash = current_hash
        self._last_result = final_result
        print(f"DEBUG: Извлечено {len(final_result)} записей из {num_pages} страниц")
        return final_result

    def _extract_from_text(self, page_text, full_text=None, page_num=1):
        if full_text is None:
            full_text = page_text

        prompt = f"""
Ты — экстрактор данных из финансовых документов. Верни ТОЛЬКО валидный JSON. Без пояснений, без markdown.

Обязательные поля (если не найдено — null):
- date: дата документа в формате ДД.ММ.ГГГГ
- total: итоговая сумма строкой, например "258.00"
- currency: код валюты — ТОЛЬКО одно из: RUB, USD, EUR, BYN
- invoice_number: номер чека, счёта, квитанции
- company: название организации, магазина или получателя платежа
- vat: сумма НДС строкой, если есть, иначе null
- document_type: тип документа одним словом или коротко, например: "Кассовый чек", "Банковская квитанция", "Счёт-фактура", "Товарный чек", "Платёжное поручение"

Дополнительное поле extra_fields — словарь с ЛЮБЫМИ важными данными которые ты нашёл в документе.
Включай в extra_fields всё значимое: плательщик, получатель, банк, назначение платежа, номер карты, номер телефона, адрес, артикулы товаров, количество позиций, скидка, бонусы, кассир, терминал, номер ЕРИП, код авторизации и т.д.
НЕ включай в extra_fields то что уже есть в основных полях (date, total, currency, invoice_number, company, vat).
Если дополнительных данных нет — extra_fields: {{}}.

Примеры:

Текст: "Чек №456 от 20.01.2024. Магазин Пятёрочка. Итого: 890.00 руб. Кассир: Анна. Товаров: 5"
Вывод: {{"date":"20.01.2024","total":"890.00","currency":"RUB","invoice_number":"456","company":"Пятёрочка","vat":null,"document_type":"Кассовый чек","extra_fields":{{"Кассир":"Анна","Количество товаров":"5"}}}}

Текст: "Приорбанк. Вывоз мусора. Плательщик: Иван Иванов. Сумма 258 BYN. Получатель: Датком Столица. Номер ЕРИП: 7788888. Код авторизации: 542121"
Вывод: {{"date":null,"total":"258.00","currency":"BYN","invoice_number":null,"company":"Датком Столица","vat":null,"document_type":"Банковская квитанция","extra_fields":{{"Плательщик":"Иван Иванов","Банк":"Приорбанк","Услуга":"Вывоз мусора","Номер ЕРИП":"7788888","Код авторизации":"542121"}}}}

Теперь извлеки из следующего текста (страница {page_num}):

{full_text}

Вывод (только JSON):
"""

        try:
            response = ollama.chat(
                model=config['llm']['model'],
                messages=[{'role': 'user', 'content': prompt}],
                options={'temperature': config['llm']['temperature']}
            )
        except Exception as e:
            logging.error(f"Ollama error on page {page_num}: {e}")
            return None

        content = response['message']['content']
        print(f"[LLM RAW OUTPUT page {page_num}]: {content[:200]}...")

        first_brace = content.find('{')
        last_brace = content.rfind('}')
        if first_brace == -1 or last_brace == -1:
            logging.error(f"No JSON found on page {page_num}")
            return None

        json_str = content[first_brace:last_brace + 1]
        json_str = clean_json_trailing_commas(json_str)

        try:
            data = json.loads(json_str)
        except json.JSONDecodeError:
            objects = []
            brace_count = 0
            start_idx = -1
            for i, ch in enumerate(content):
                if ch == '{':
                    if brace_count == 0:
                        start_idx = i
                    brace_count += 1
                elif ch == '}':
                    brace_count -= 1
                    if brace_count == 0 and start_idx != -1:
                        obj_str = content[start_idx:i + 1]
                        try:
                            obj_str_cleaned = clean_json_trailing_commas(obj_str)
                            objects.append(json.loads(obj_str_cleaned))
                        except:
                            pass
                        start_idx = -1
            if objects:
                data = objects[0]
            else:
                logging.error(f"Could not parse JSON on page {page_num}")
                return None

        try:
            validated = InvoiceData(**data)
            result = validated.dict()

            # Фолбэки если LLM не нашёл — ищем регулярками
            if not result.get('invoice_number'):
                match = re.search(r'(?:№|Номер)\s*[:]?\s*([0-9\-]+)', full_text)
                if match:
                    result['invoice_number'] = match.group(1)

            if not result.get('company'):
                patterns = [
                    r'(?:Поставщик|Исполнитель|Продавец|Получатель платежа)\s*[:]?\s*([А-Яа-я0-9\s\-\"\'\,\(\)]+?)(?:\n|,|$)',
                    r'(?:ООО|ЗАО|АО|ИП|ЧУП|ОАО|ПАО)\s+[А-Яа-я0-9\s\-\"\']+',
                ]
                for pat in patterns:
                    match = re.search(pat, full_text, re.IGNORECASE)
                    if match:
                        company_candidate = match.group(1) if match.lastindex else match.group(0)
                        company_candidate = company_candidate.strip().strip('"\'')
                        if len(company_candidate) > 2:
                            result['company'] = company_candidate
                            break

            if not result.get('vat'):
                match = re.search(r'НДС\s*[:]?\s*([0-9\.,]+)', full_text, re.IGNORECASE)
                if match:
                    vat_str = match.group(1).replace(',', '.')
                    result['vat'] = vat_str

            return result

        except Exception as e:
            logging.error(f"Validation error on page {page_num}: {e}")
            return None

    def interpret(self, ocr_json):
        result = self.interpret_multi(ocr_json)
        if result and len(result) > 0:
            return result[0]
        return {"error": "No data extracted"}

from datetime import datetime


class ValidationAgent:
    def validate(self, data: dict) -> dict:
        errors = []

        if not self.check_total(data):
            errors.append("Сумма товаров не совпадает с итоговой суммой")

        if not self.check_currency(data):
            errors.append("Неподдерживаемая валюта")

        if not self.check_date(data):
            errors.append("Неверная дата")

        if not self.check_negative_prices(data):
            errors.append("Есть отрицательная цена")

        return {
            "valid": len(errors) == 0,
            "errors": errors
        }

    def check_total(self, data: dict) -> bool:
        items = data.get("items", [])
        total = data.get("total")

        if total is None:
            return False

        if not items:
            return True

        try:
            calc_sum = sum(float(item.get("price", 0)) for item in items if isinstance(item, dict))
            return abs(calc_sum - float(total)) < 0.01
        except Exception:
            return False

    def check_currency(self, data: dict) -> bool:
        currency = str(data.get("currency", "")).upper().strip()
        return currency in ["BYN", "RUB", "USD", "EUR", ""]

    def check_date(self, data: dict) -> bool:
        date_str = str(data.get("date", "")).strip()

        if not date_str or date_str == "N/A" or date_str == "None" or date_str == "":
            return True

        formats = [
            "%d.%m.%Y",
            "%d.%m.%Y %H:%M:%S",
            "%Y-%m-%d",
            "%Y-%m-%d %H:%M:%S",
            "%d/%m/%Y",
            "%d/%m/%Y %H:%M:%S",
        ]

        for fmt in formats:
            try:
                datetime.strptime(date_str, fmt)
                return True
            except:
                continue
        return False

    def check_negative_prices(self, data: dict) -> bool:
        items = data.get("items", [])
        if not items:
            return True

        try:
            return all(float(item.get("price", 0)) >= 0 for item in items if isinstance(item, dict))
        except Exception:
            return False

    def _extract_category(self, response) -> str:
        raw_category = ""

        if isinstance(response, dict):
            raw_category = str(response.get("category", "")).strip()
        elif isinstance(response, str):
            raw_category = response.strip()
        else:
            raw_category = str(response).strip()

        return self._normalize_category(raw_category)

    def _normalize_category(self, value: str) -> str:
        if not value:
            return "прочее"

        key = value.lower().strip().replace("ё", "е")
        key = key.replace("_", " ").replace("-", " ")

        if key in self.CATEGORY_ALIASES:
            return self.CATEGORY_ALIASES[key]

        if value in self.ALLOWED_CATEGORIES:
            return value

        return "прочее"

async function addFiles(fileList) {
  for (const file of fileList) {
    if (!file.type.match(/image\/(png|jpeg)|application\/pdf/)) {
      showToast('Неподдерживаемый формат: ' + file.name, 'error'); continue;
    }
    const pages = await extractPages(file);
    state.files.push({ file, name: file.name, pages, status: 'idle', result: null });
  }
  if (state.files.length > 0) { state.currentFileIdx = state.files.length - 1; state.currentPageIdx = 0; }
  renderFileList(); showPreview(); updateButtons(); updateNavButtons();
}

async function extractPages(file) {
  if (file.type !== 'application/pdf') return [{ blob: URL.createObjectURL(file), rotation: 0 }];
  try { return await renderPDFPages(file); } catch(e) { return [{ blob: URL.createObjectURL(file), rotation: 0 }]; }
}

async function renderPDFPages(file) {
  if (!window.pdfjsLib) {
    await loadScript('https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.min.js');
    window.pdfjsLib.GlobalWorkerOptions.workerSrc =
      'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';
  }
  const ab = await file.arrayBuffer();
  const pdf = await window.pdfjsLib.getDocument({ data: ab }).promise;
  const pages = [];
  for (let i = 1; i <= pdf.numPages; i++) {
    const page = await pdf.getPage(i);
    const vp = page.getViewport({ scale: 1.5 });
    const canvas = document.createElement('canvas');
    canvas.width = vp.width; canvas.height = vp.height;
    await page.render({ canvasContext: canvas.getContext('2d'), viewport: vp }).promise;
    pages.push({ blob: canvas.toDataURL('image/png'), rotation: 0 });
  }
  return pages;
}

function loadScript(src) {
  return new Promise((res, rej) => {
    const s = document.createElement('script'); s.src = src; s.onload = res; s.onerror = rej;
    document.head.appendChild(s);
  });
}

function renderFileList() {
  const list = document.getElementById('fileList');
  list.innerHTML = '';
  state.files.forEach((f, i) => {
    const div = document.createElement('div');
    div.className = 'file-item' + (i === state.currentFileIdx ? ' active' : '');
    div.onclick = () => selectFile(i);
    const isPDF = f.file.type === 'application/pdf';
    const firstBlob = f.pages[0] && f.pages[0].blob;
    const thumbHTML = (!isPDF && firstBlob) ? `<img src="${firstBlob}" alt="">` : '📄';
    const statusColor = {idle:'var(--muted)',processing:'var(--amber)',done:'var(--green)',error:'var(--red)'}[f.status];
    const statusIcon  = {idle:'○',processing:'⟳',done:'✓',error:'✗'}[f.status];
    div.innerHTML = `
      <div class="file-thumb">${thumbHTML}</div>
      <div class="file-info">
        <div class="file-name" title="${escHtml(f.name)}">${escHtml(f.name)}</div>
        <div class="file-meta">${f.pages.length} стр. &nbsp;<span style="color:${statusColor}">${statusIcon}</span></div>
      </div>
      <button class="file-remove" onclick="removeFile(event,${i})">×</button>`;
    list.appendChild(div);
  });
  document.getElementById('fileCountBadge').textContent = state.files.length + ' файлов';
}

function selectFile(idx) {
  state.currentFileIdx = idx; state.currentPageIdx = 0;
  renderFileList(); showPreview(); updateNavButtons();
  clearHighlights();
}

function removeFile(e, idx) {
  e.stopPropagation(); state.files.splice(idx, 1);
  if (state.currentFileIdx >= state.files.length) state.currentFileIdx = Math.max(0, state.files.length - 1);
  state.currentPageIdx = 0;
  renderFileList(); showPreview(); updateButtons(); updateNavButtons(); clearHighlights();
}

function showPreview() {
  const empty = document.getElementById('previewEmpty');
  const wrap  = document.getElementById('previewImgWrap');
  const img   = document.getElementById('previewImg');

  if (state.files.length === 0) {
    empty.style.display = ''; wrap.classList.remove('visible');
    document.getElementById('previewCounter').textContent = '—';
    document.getElementById('previewFileCounter').textContent = '—';
    document.getElementById('previewPageCounter').textContent = '—';
    document.getElementById('previewFileName').textContent = 'Нет файла';
    return;
  }
  const f = state.files[state.currentFileIdx];
  if (!f) return;
  const page = f.pages[state.currentPageIdx];
  if (!page) return;

  empty.style.display = 'none'; wrap.classList.add('visible');
  img.style.transform = `rotate(${page.rotation || 0}deg)`;
  img.src = page.blob;

  const fi = state.currentFileIdx + 1, ft = state.files.length;
  const pi = state.currentPageIdx + 1,  pt = f.pages.length;
  document.getElementById('previewCounter').textContent = `${fi}/${ft} · стр.${pi}/${pt}`;
  document.getElementById('previewFileCounter').textContent = `${fi} / ${ft}`;
  document.getElementById('previewPageCounter').textContent = `стр.${pi}/${pt}`;
  document.getElementById('previewFileName').textContent = f.name;

  clearHighlights();
  // Rebuild OCR hit zones if we have results for this file/page
  rebuildOcrHitLayer();
}

// Called when the preview <img> finishes loading (natural size becomes known)
function onPreviewImageLoad() {
  const img = document.getElementById('previewImg');
  state.imgNaturalW = img.naturalWidth;
  state.imgNaturalH = img.naturalHeight;
  state.imgDisplayW = img.offsetWidth;
  state.imgDisplayH = img.offsetHeight;
  // Resize canvas to match image display size
  const canvas = document.getElementById('highlightCanvas');
  canvas.width  = img.offsetWidth;
  canvas.height = img.offsetHeight;
  rebuildOcrHitLayer();
}

// Update canvas size on window resize
window.addEventListener('resize', () => {
  const img = document.getElementById('previewImg');
  if (!img.src || img.src === window.location.href) return;
  state.imgDisplayW = img.offsetWidth;
  state.imgDisplayH = img.offsetHeight;
  const canvas = document.getElementById('highlightCanvas');
  canvas.width  = img.offsetWidth;
  canvas.height = img.offsetHeight;
  redrawCanvas();
});

function navigateFile(dir) {
  const newIdx = state.currentFileIdx + dir;
  if (newIdx < 0 || newIdx >= state.files.length) return;
  state.currentFileIdx = newIdx; state.currentPageIdx = 0;
  renderFileList(); showPreview(); updateNavButtons(); clearHighlights();
  if (state.results.length > 0) {
    state.currentResFileIdx = Math.min(newIdx, state.results.length - 1);
    state.currentResPageIdx = 0; renderResult();
  }
}

function navigatePage(dir) {
  if (state.files.length === 0) return;
  const f = state.files[state.currentFileIdx];
  const newIdx = state.currentPageIdx + dir;
  if (newIdx < 0 || newIdx >= f.pages.length) return;
  state.currentPageIdx = newIdx; showPreview(); updateNavButtons(); clearHighlights();
  const res = state.results[state.currentResFileIdx];
  if (res && newIdx < res.parsed_list.length) { state.currentResPageIdx = newIdx; renderResult(); }
}

function rotatePreview() {
  if (state.files.length === 0) return;
  const page = state.files[state.currentFileIdx].pages[state.currentPageIdx];
  if (!page) return;
  page.rotation = ((page.rotation || 0) + 90) % 360;
  showPreview();
}

function updateNavButtons() {
  const f = state.files.length > 0 ? state.files[state.currentFileIdx] : null;
  document.getElementById('prevFileBtn').disabled = state.currentFileIdx <= 0;
  document.getElementById('nextFileBtn').disabled = state.currentFileIdx >= state.files.length - 1;
  document.getElementById('prevPageBtn').disabled = state.currentPageIdx <= 0;
  document.getElementById('nextPageBtn').disabled = !f || state.currentPageIdx >= f.pages.length - 1;
}

function updateButtons() {
  const has = state.files.length > 0;
  document.getElementById('btnProcess').disabled = !has || state.processing;
  document.getElementById('btnRotate').disabled  = !has;
  document.getElementById('btnClear').disabled   = !has;
  updateNavButtons();
}

function clearAll() {
  state.files = []; state.currentFileIdx = 0; state.currentPageIdx = 0;
  state.results = []; state.currentResFileIdx = 0; state.currentResPageIdx = 0;
  renderFileList(); showPreview(); updateButtons(); clearHighlights();
  document.getElementById('resultEmpty').style.display = '';
  document.getElementById('resultEmpty').innerHTML = `
    <div class="result-empty-icon">📊</div>
    <div style="font-size:13px;color:var(--muted)">Запустите анализ</div>
    <div style="font-size:11px;color:var(--muted);margin-top:4px">Результаты появятся здесь</div>`;
  document.getElementById('resultContent').style.display = 'none';
  document.getElementById('resultNav').style.display = 'none';
  document.getElementById('actionsRow').style.display = 'none';
  clearLog(); setStatus('idle', 'Готов');
}

// Field colours (consistent across both panels)
const FIELD_COLORS = {
  total:          { fill: 'rgba(79,142,247,0.25)',  stroke: 'rgba(79,142,247,0.9)',  label: 'Сумма' },
  date:           { fill: 'rgba(45,212,160,0.25)',  stroke: 'rgba(45,212,160,0.9)',  label: 'Дата' },
  company:        { fill: 'rgba(245,166,35,0.25)',  stroke: 'rgba(245,166,35,0.9)',  label: 'Компания' },
  invoice_number: { fill: 'rgba(124,92,252,0.25)',  stroke: 'rgba(124,92,252,0.9)',  label: 'Номер счёта' },
  vat:            { fill: 'rgba(247,111,111,0.25)', stroke: 'rgba(247,111,111,0.9)', label: 'НДС' },
  currency:       { fill: 'rgba(45,212,160,0.18)',  stroke: 'rgba(45,212,160,0.6)',  label: 'Валюта' },
};

// Цвет для extra-полей (все одним цветом — жёлтый)
const EXTRA_COLOR = { fill: 'rgba(245,166,35,0.2)', stroke: 'rgba(245,166,35,0.85)', label: '' };

function getFieldColor(key) {
  if (key.startsWith('extra::')) return EXTRA_COLOR;
  return FIELD_COLORS[key] || { fill: 'rgba(255,255,255,0.15)', stroke: 'rgba(255,255,255,0.6)', label: key };
}

// Scale OCR pixel coords → canvas coords
function scaleBlock(block) {
  const scaleX = state.imgDisplayW / (state.imgNaturalW || 1);
  const scaleY = state.imgDisplayH / (state.imgNaturalH || 1);
  return {
    x: block.left   * scaleX,
    y: block.top    * scaleY,
    w: block.width  * scaleX,
    h: block.height * scaleY,
  };
}

function drawBlock(ctx, block, fieldName) {
  const color  = getFieldColor(fieldName);
  const scaled = scaleBlock(block);
  ctx.fillStyle   = color.fill;
  ctx.strokeStyle = color.stroke;
  ctx.lineWidth   = 2;
  // Rounded rect
  const r = 3;
  ctx.beginPath();
  ctx.moveTo(scaled.x + r, scaled.y);
  ctx.lineTo(scaled.x + scaled.w - r, scaled.y);
  ctx.quadraticCurveTo(scaled.x + scaled.w, scaled.y, scaled.x + scaled.w, scaled.y + r);
  ctx.lineTo(scaled.x + scaled.w, scaled.y + scaled.h - r);
  ctx.quadraticCurveTo(scaled.x + scaled.w, scaled.y + scaled.h, scaled.x + scaled.w - r, scaled.y + scaled.h);
  ctx.lineTo(scaled.x + r, scaled.y + scaled.h);
  ctx.quadraticCurveTo(scaled.x, scaled.y + scaled.h, scaled.x, scaled.y + scaled.h - r);
  ctx.lineTo(scaled.x, scaled.y + r);
  ctx.quadraticCurveTo(scaled.x, scaled.y, scaled.x + r, scaled.y);
  ctx.closePath();
  ctx.fill();
  ctx.stroke();

  // Small label tag
  const color2 = getFieldColor(fieldName);
  const labelText = color2.label || (fieldName.startsWith('extra::') ? fieldName.replace('extra::', '') : fieldName);
  if (labelText) {
    ctx.fillStyle = color2.stroke;
    ctx.font = 'bold 9px IBM Plex Mono, monospace';
    ctx.fillText(labelText, scaled.x + 2, scaled.y - 2);
  }
}

function redrawCanvas() {
  const canvas = document.getElementById('highlightCanvas');
  const ctx    = canvas.getContext('2d');
  ctx.clearRect(0, 0, canvas.width, canvas.height);

  if (!state.activeField && !state.activeBlock) return;

  // Determine which field to highlight
  let fieldName = null;
  if (state.activeField) fieldName = state.activeField.fieldName;
  else if (state.activeBlock) fieldName = state.activeBlock.fieldName;

  if (!fieldName) return;

  // Get blocks for current file/page
  const mapping = getCurrentMapping();
  if (!mapping) return;
  const blocks = mapping[fieldName];
  if (!blocks || blocks.length === 0) return;

  blocks.forEach(block => drawBlock(ctx, block, fieldName));
  document.getElementById('hlLegend').classList.add('visible');
}

function clearHighlights() {
  state.activeField = null; state.activeBlock = null;
  const canvas = document.getElementById('highlightCanvas');
  const ctx    = canvas.getContext('2d');
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  document.getElementById('hlLegend').classList.remove('visible');
  // Clear result field highlights
  document.querySelectorAll('.result-field.hl-active').forEach(el => el.classList.remove('hl-active'));
}

function getCurrentMapping() {
  const res = state.results[state.currentResFileIdx];
  if (!res || !res.field_mappings) return null;
  return res.field_mappings[state.currentResPageIdx] || null;
}

function getResultForFileIdx(fileIdx) {
  return state.results[fileIdx] || null;
}

// ── Hover from result field → highlight on canvas ──────────────
function onResultFieldEnter(fieldName) {
  state.activeField = { fieldName };
  state.activeBlock = null;
  redrawCanvas();
}

function onResultFieldLeave() {
  state.activeField = null;
  redrawCanvas();
}

// ── Hover from OCR hit zone → highlight result field ───────────
function onOcrBlockEnter(fieldName) {
  state.activeBlock = { fieldName };
  state.activeField = null;
  redrawCanvas();
  // Highlight the corresponding result field
  document.querySelectorAll('.result-field.hl-active').forEach(el => el.classList.remove('hl-active'));
  const el = document.querySelector(`.result-field[data-field="${fieldName}"]`);
  if (el) {
    el.classList.add('hl-active');
    el.scrollIntoView({ block: 'nearest', behavior: 'smooth' });
  }
}

function onOcrBlockLeave() {
  state.activeBlock = null;
  redrawCanvas();
  document.querySelectorAll('.result-field.hl-active').forEach(el => el.classList.remove('hl-active'));
}

// Build invisible hit divs over each OCR block that has a field mapping
function rebuildOcrHitLayer() {
  const layer = document.getElementById('ocrHitLayer');
  layer.innerHTML = '';

  const mapping = getCurrentMapping();
  if (!mapping) return;

  // Recompute display size (image may have just loaded)
  const img = document.getElementById('previewImg');
  if (!img.complete || !img.naturalWidth) return;
  state.imgNaturalW = img.naturalWidth;
  state.imgNaturalH = img.naturalHeight;
  state.imgDisplayW = img.offsetWidth;
  state.imgDisplayH = img.offsetHeight;

  Object.entries(mapping).forEach(([fieldName, blocks]) => {
    blocks.forEach(block => {
      const scaled = scaleBlock(block);
      const div = document.createElement('div');
      div.className = 'ocr-hit';
      div.style.left   = scaled.x + 'px';
      div.style.top    = scaled.y + 'px';
      div.style.width  = scaled.w + 'px';
      div.style.height = scaled.h + 'px';
      const fc = getFieldColor(fieldName);
      div.title = fc.label || (fieldName.startsWith('extra::') ? fieldName.replace('extra::', '') : fieldName);
      div.addEventListener('mouseenter', () => onOcrBlockEnter(fieldName));
      div.addEventListener('mouseleave', () => onOcrBlockLeave());
      layer.appendChild(div);
    });
  });
}

// ════════════════════════════════════════════════
// PROCESSING
// ════════════════════════════════════════════════
async function startProcessing() {
  if (state.files.length === 0 || state.processing) return;

  // Только файлы которые ещё не обработаны (idle или error)
  const toProcess = state.files.filter(f => f.status !== 'done');
  if (toProcess.length === 0) {
    showToast('Все файлы уже обработаны', 'info'); return;
  }

  state.processing = true;
  // НЕ сбрасываем state.results — сохраняем уже готовые результаты
  // Убеждаемся что results выровнен по files (заполняем null для уже готовых)
  while (state.results.length < state.files.length) state.results.push(null);

  setStatus('processing', 'Обработка...');
  document.getElementById('btnProcess').disabled = true;
  document.getElementById('processingOverlay').classList.add('active');
  clearLog();

  let newCount = 0;

  for (let i = 0; i < state.files.length; i++) {
    const f = state.files[i];

    // Пропускаем уже обработанные
    if (f.status === 'done') {
      addLog(`⏩ Пропущен (уже обработан): ${f.name}`, 'info');
      continue;
    }

    f.status = 'processing'; renderFileList();
    addLog(`Файл ${i+1}/${state.files.length}: ${f.name}`, 'step');
    document.getElementById('processingStep').textContent = `${i+1}/${state.files.length}: ${f.name}`;

    try {
      const result = await processFile(f);
      f.status = 'done'; f.result = result;
      state.results[i] = result;  // сохраняем по индексу, не push
      if (result.logs) result.logs.forEach(l =>
        addLog(l, l.includes('ОШИБКА')||l.includes('✗') ? 'error' : l.includes('ШАГ') ? 'step' : l.includes('ЗАВЕРШЕНА') ? 'success' : 'info')
      );
      addLog(`✓ ${f.name}: ${result.parsed_list.length} записей`, 'success');
      newCount++;
    } catch(e) {
      f.status = 'error';
      state.results[i] = { filename: f.name, parsed_list: [], logs: [], pages_blocks: [], field_mappings: [], error: String(e) };
      addLog(`✗ ${f.name}: ${e}`, 'error');
    }
    renderFileList();
  }

  state.processing = false;
  document.getElementById('processingOverlay').classList.remove('active');
  document.getElementById('btnProcess').disabled = false;

  // Показываем результаты начиная с первого нового файла (или первого вообще)
  const firstNewIdx = state.files.findIndex(f => f.status === 'done' && state.results[state.files.indexOf(f)]);
  if (firstNewIdx !== -1) {
    state.currentResFileIdx = firstNewIdx;
    state.currentResPageIdx = 0;
  }

  const hasResults = state.results.some(r => r && r.parsed_list && r.parsed_list.length > 0);
  if (hasResults) {
    document.getElementById('resultEmpty').style.display = 'none';
    document.getElementById('resultContent').style.display = 'block';
    document.getElementById('resultNav').style.display = 'flex';
    document.getElementById('actionsRow').style.display = 'flex';
    renderResult();
    rebuildOcrHitLayer();
  }

  const doneCount = state.results.filter(r => r && r.parsed_list && r.parsed_list.length > 0).length;
  setStatus(doneCount > 0 ? 'done' : 'error', doneCount > 0 ? 'Завершено' : 'Ошибки');
  showToast(`Обработано новых: ${newCount}, всего готово: ${doneCount}/${state.files.length}`, doneCount > 0 ? 'success' : 'error');
  loadHistory().catch(()=>{});
}

async function processFile(fileObj) {
  const formData = new FormData();
  formData.append('file', fileObj.file, fileObj.name);
  const r = await fetch('/api/process', { method: 'POST', body: formData });
  if (!r.ok) { const err = await r.json().catch(()=>({})); throw new Error(err.error || `HTTP ${r.status}`); }
  return await r.json();
}

function navigateResultFile(dir) {
  const newIdx = state.currentResFileIdx + dir;
  if (newIdx < 0 || newIdx >= state.results.length) return;
  state.currentResFileIdx = newIdx; state.currentResPageIdx = 0;
  clearHighlights();
  if (newIdx < state.files.length) {
    state.currentFileIdx = newIdx; state.currentPageIdx = 0;
    renderFileList(); showPreview(); updateNavButtons();
  }
  renderResult(); rebuildOcrHitLayer();
}

function navigateResultPage(dir) {
  const res = state.results[state.currentResFileIdx];
  if (!res) return;
  const newIdx = state.currentResPageIdx + dir;
  if (newIdx < 0 || newIdx >= res.parsed_list.length) return;
  state.currentResPageIdx = newIdx; clearHighlights();
  const f = state.files[state.currentFileIdx];
  if (f && newIdx < f.pages.length) { state.currentPageIdx = newIdx; showPreview(); updateNavButtons(); }
  renderResult(); rebuildOcrHitLayer();
}

function renderResult() {
  const res   = state.results[state.currentResFileIdx];
  const empty = document.getElementById('resultEmpty');
  const cont  = document.getElementById('resultContent');

  if (!res) { empty.style.display=''; cont.style.display='none'; return; }

  if (res.error && res.parsed_list.length === 0) {
    empty.style.display=''; cont.style.display='none';
    empty.innerHTML=`<div class="result-empty-icon" style="opacity:1">⚠</div><div style="font-size:13px;color:var(--red)">${escHtml(res.error)}</div>`;
    document.getElementById('resultCounter').textContent='—'; return;
  }
  if (res.parsed_list.length === 0) {
    empty.style.display=''; cont.style.display='none';
    empty.innerHTML=`<div class="result-empty-icon">🔍</div><div style="font-size:13px;color:var(--muted)">Данные не найдены</div>`; return;
  }

  empty.style.display='none'; cont.style.display='block';

  const parsed = res.parsed_list[state.currentResPageIdx];
  const fi = state.currentResFileIdx+1, ft = state.results.length;
  const pi = state.currentResPageIdx+1, pt = res.parsed_list.length;
  const mapping = getCurrentMapping() || {};

  document.getElementById('resultCounter').textContent = `${fi}/${ft} · стр.${pi}/${pt}`;
  document.getElementById('resultFileCounter').textContent = `${fi} / ${ft}`;
  document.getElementById('resultPageCounter').textContent = `стр.${pi}/${pt}`;
  document.getElementById('resultFileName').textContent = res.filename || '—';
  document.getElementById('prevResFileBtn').disabled = state.currentResFileIdx <= 0;
  document.getElementById('nextResFileBtn').disabled = state.currentResFileIdx >= state.results.length-1;
  document.getElementById('prevResPageBtn').disabled = state.currentResPageIdx <= 0;
  document.getElementById('nextResPageBtn').disabled = state.currentResPageIdx >= pt-1;

  const isValid = parsed.validation_valid;
  const errors  = parsed.validation_errors || [];
  const category = parsed.category || 'прочее';
  const docType  = parsed.document_type || null;
  const extraFields = parsed.extra_fields || {};

  // ── Иконки по типу документа ──
  const docTypeIcons = {
    'кассовый чек': '🧾', 'товарный чек': '🛍', 'банковская квитанция': '🏦',
    'квитанция': '🏦', 'счёт-фактура': '📋', 'счёт': '📄',
    'платёжное поручение': '💳', 'накладная': '📦', 'акт': '📝',
  };
  const docIcon = docType
    ? (docTypeIcons[docType.toLowerCase()] || '📄')
    : '📄';

  // ── Базовые поля (только непустые) ──
  const coreFields = [
    { key: 'total',          label: 'Итого',        cls: 'big', suffix: ` ${escHtml(parsed.currency||'RUB')}` },
    { key: 'date',           label: 'Дата',          cls: '' },
    { key: 'company',        label: 'Компания',      cls: '' },
    { key: 'invoice_number', label: 'Номер',         cls: '' },
    { key: 'vat',            label: 'НДС',           cls: '' },
  ];

  function buildFieldRow(key, label, val, cls, suffix) {
    if (!val) return '';
    const hasMapping = mapping[key] && mapping[key].length > 0;
    const dot = hasMapping
      ? `<span style="display:inline-block;width:6px;height:6px;border-radius:50%;background:${FIELD_COLORS[key]?.stroke||'#fff'};margin-right:5px;flex-shrink:0;margin-bottom:1px"></span>`
      : '';
    const hover = hasMapping
      ? `onmouseenter="onResultFieldEnter('${key}')" onmouseleave="onResultFieldLeave()"`
      : '';
    const tooltip = hasMapping
      ? `<span class="field-tooltip" style="color:${FIELD_COLORS[key]?.stroke}">↔ подсветить</span>`
      : '';
    return `
      <div class="result-field" data-field="${key}" ${hover}>
        <span class="result-field-key">${dot}${escHtml(label)}</span>
        <span class="result-field-val ${cls}">${escHtml(String(val))}${suffix ? `<span style="font-size:12px;color:var(--muted);margin-left:4px">${suffix}</span>` : ''}</span>
        ${tooltip}
      </div>`;
  }

  const coreRows = coreFields
    .map(f => buildFieldRow(f.key, f.label, parsed[f.key], f.cls, f.suffix||''))
    .filter(Boolean)
    .join('');

  // ── Дополнительные поля (extra_fields) ──
  const extraRows = Object.entries(extraFields).map(([k, v]) => {
    const mapKey = `extra::${k}`;
    const hasMap = mapping[mapKey] && mapping[mapKey].length > 0;
    const dot = hasMap
      ? `<span style="display:inline-block;width:6px;height:6px;border-radius:50%;background:rgba(245,166,35,0.9);margin-right:5px;flex-shrink:0;margin-bottom:1px"></span>`
      : '';
    const hover = hasMap
      ? `onmouseenter="onResultFieldEnter('${mapKey}')" onmouseleave="onResultFieldLeave()"`
      : '';
    const tooltip = hasMap
      ? `<span class="field-tooltip" style="color:rgba(245,166,35,0.9)">↔ подсветить</span>`
      : '';

def preprocess_image(file_path):
    base_dir = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
    data_dir = os.path.join(base_dir, "data")

    if not os.path.exists(data_dir):
        os.makedirs(data_dir)

    img = Image.open(file_path)

    img = img.convert("L")

    img = ImageEnhance.Contrast(img).enhance(3.0)

    img = ImageEnhance.Sharpness(img).enhance(2.0)


    threshold = 140
    img = img.point(lambda p: 255 if p > threshold else 0)
    img = img.convert("L")

    save_path = os.path.join(data_dir, "temp_proc.png")
    img.save(save_path)
    return save_path

import re
from deep_translator import GoogleTranslator


def normalize_data(text):
    if not text:
        return ""

    # === НАЧАЛО БЛОКА: ИСПРАВЛЕНИЕ OCR-ОШИБОК ВАЛЮТ ===
    # Очищаем строку от пробелов и переводим в верхний регистр для точного сравнения
    cleaned_currency = str(text).strip().upper()
    
    # Если Tesseract перепутал буквы и прочитал BYN как ВУМ/VUM/БУН/ВУН, 
    # мы возвращаем чистый, правильный ISO-код BYN
    if cleaned_currency in ['ВУМ', 'VUM', 'БУН', 'BYM', 'ВУН', 'BYN']:
        return 'BYN'
        
    # То же самое делаем для других стандартных валют на случай лишних пробелов
    if cleaned_currency in ['RUB', 'РУБ', 'РУБ.']:
        return 'RUB'
    if cleaned_currency in ['USD', 'EUR', 'KZT']:
        return cleaned_currency
    # === КОНЕЦ БЛОКА ===

    # ТВОЙ СТАРЫЙ КОД (ОСТАЛСЯ ПОЛНОСТЬЮ БЕЗ ИЗМЕНЕНИЙ)
    text = text.replace("долларов США", "$").replace("доллара", "$").replace("доллар", "$")
    text = text.replace("евро", "€")

    months = {
        "января": "01", "февраля": "02", "марта": "03", "апреля": "04",
        "мая": "05", "июня": "06", "июля": "07", "августа": "08",
        "сентября": "09", "октября": "10", "ноября": "11", "декабря": "12",
    }
    for name, num in months.items():
        text = re.sub(rf"(\d{{1,2}})\s{name}\s(\d{{4}})", rf"\1.{num}.\2", text, flags=re.IGNORECASE)

    date_pattern = r"(\d{2,4})[-/](\d{2})[-/](\d{2,4})"

    def date_fixer(match):
        g1, g2, g3 = match.groups()
        if len(g1) == 4:
            return f"{g3}.{g2}.{g1}"
        return f"{g1}.{g2}.{g3}"

    return re.sub(date_pattern, date_fixer, text)


def translate_field(text):
    print(r"=== ВХОД В ПЕРЕВОДЧИК ===", repr(text))  # Покажет, что прислал ИИ
    
    if not text:
        return ""
        
    cleaned_text = str(text).strip().upper()
    ISO_CURRENCIES = ['USD', 'EUR', 'BYN', 'RUB', 'KZT', 'UAH']
    
    if cleaned_text in ISO_CURRENCIES:
        print(r"=== СРАБОТАЛ ФИЛЬТР ВАЛЮТ ===", cleaned_text)
        return cleaned_text

    try:
        translated = GoogleTranslator(source="auto", target="ru").translate(text)
        res = normalize_data(translated)
        print(r"=== ВЫХОД ИЗ ПЕРЕВОДЧИКА (УСПЕХ) ===", repr(res))
        return res
    except:
        res = normalize_data(text)
        print(r"=== ВЫХОД ИЗ ПЕРЕВОДЧИКА (ОШИБКА) ===", repr(res))
        return res

class Orchestrator:
    def __init__(self, ocr_module, llm_agent, db_module, classifier, validator,
                 translator_func=None):
        self.ocr = ocr_module
        self.llm = llm_agent
        self.db = db_module
        self.classifier = classifier
        self.validator = validator
        self.translator = translator_func

    def process_document(self, file_path, filename):
        logs = []
        doc_id = None

        try:
            logs.append(f"НАЧАЛО ОБРАБОТКИ: {filename}")

            logs.append("ШАГ 1: Предобработка изображения...")
            processed_path = preprocess_image(file_path) if file_path.lower().endswith(
                ('.png', '.jpg', '.jpeg')) else file_path
            logs.append("Предобработка завершена")

            logs.append("ШАГ 2: Распознавание текста (OCR)...")
            ocr_data = self.ocr.extract(processed_path if processed_path else file_path)

            full_ocr_text = "\n\n--- РАЗРЫВ СТРАНИЦЫ ---\n\n".join(
                page['text'] for page in ocr_data['pages']
            )

            logs.append(f"OCR завершён. Символов: {len(full_ocr_text)}")
            logs.append(f"OCR движок: {ocr_data['ocr_engine']}")
            logs.append(f"Страниц: {len(ocr_data['pages'])}")

            logs.append("ШАГ 3: Нормализация текста (даты, валюты)...")
            normalized_text = normalize_data(full_ocr_text)

            translated_text = normalized_text
            if self.translator:
                logs.append("ШАГ 4: Перевод на русский...")
                translated_text = self.translator(normalized_text)
                logs.append("Перевод выполнен")
            else:
                logs.append("ШАГ 4: Перевод отключён")

            original_pages_text = [page['text'] for page in ocr_data['pages']]
            for i, page in enumerate(ocr_data['pages']):
                if self.translator:
                    page['text'] = self.translator(page['text'])
                else:
                    page['text'] = normalize_data(page['text'])

            logs.append("ШАГ 5: LLM интерпретация (извлечение полей)...")
            parsed_list = self.llm.interpret_multi(ocr_data)
            logs.append(f"Извлечено записей: {len(parsed_list)}")

            if not parsed_list:
                raise ValueError("Не удалось извлечь данные из документа")

            for i, page in enumerate(ocr_data['pages']):
                page['text'] = original_pages_text[i]

            logs.append("ШАГ 6: Классификация и валидация...")
            enriched_parsed_list = []
            for idx, parsed in enumerate(parsed_list):
                data_for_classifier = {
                    "merchant": parsed.get('company', ''),
                    "items": [],
                    "total": parsed.get('total', 0),
                    "currency": parsed.get('currency', 'RUB'),
                    "date": parsed.get('date', '')
                }
                category_result = self.classifier.classify(data_for_classifier)

                if config['app'].get('enable_validation', True):
                    total_value = 0.0
                    if parsed.get('total'):
                        try:
                            total_str = str(parsed.get('total')).replace(',', '.')
                            numbers = re.findall(r'[\d\.]+', total_str)
                            if numbers:
                                total_value = float(numbers[0])
                        except:
                            total_value = 0.0

                    date_value = parsed.get('date', '') or ''
                    currency_value = parsed.get('currency', 'RUB') or 'RUB'

                    validation_data = {
                        "items": [],
                        "total": total_value,
                        "currency": currency_value,
                        "date": date_value
                    }
                    validation = self.validator.validate(validation_data)
                else:
                    validation = {"valid": True, "errors": []}

                enriched = parsed.copy()
                enriched['category'] = category_result.get('category', 'прочее')
                enriched['validation_valid'] = validation['valid']
                if validation.get('errors'):
                    enriched['validation_errors'] = validation['errors']
                enriched_parsed_list.append(enriched)

                logs.append(
                    f"  Запись {idx + 1}: категория={enriched['category']}, валидация={'ПРОЙДЕНА' if validation['valid'] else 'НЕ ПРОЙДЕНА'}")

            logs.append("ШАГ 7: Сохранение в базу данных...")
            ocr_version = ocr_data['ocr_engine']
            llm_version = config['llm'].get('version', config['llm']['model'])

            doc_id = self.db.save(
                filename=filename,
                ocr_text=full_ocr_text,
                parsed_data=enriched_parsed_list,
                ocr_version=ocr_version,
                llm_version=llm_version
            )

            self.db.log_event(doc_id, "OCR_COMPLETE", f"Engine: {ocr_version}")
            self.db.log_event(doc_id, "LLM_PARSE", f"Извлечено {len(parsed_list)} записей")
            self.db.log_event(doc_id, "MULTI_PAGE", f"Документ содержит {len(parsed_list)} страниц/счетов")

            for log_msg in logs:
                self.db.log_event(doc_id, "ОРКЕСТРАТОР", log_msg)

            logs.append(f"Сохранено в БД с ID: {doc_id}")
            logs.append("ОБРАБОТКА ЗАВЕРШЕНА УСПЕШНО")

            return {
                "doc_id": doc_id,
                "filename": filename,
                "parsed_list": enriched_parsed_list,
                "logs": logs,
                "ocr_text_preview": full_ocr_text[:500]
            }

        except Exception as e:
            error_msg = f"КРИТИЧЕСКАЯ ОШИБКА: {str(e)}"
            logs.append(error_msg)
            logging.error(error_msg)

            if doc_id:
                self.db.log_event(doc_id, "ОШИБКА", error_msg)

            return {
                "doc_id": None,
                "filename": filename,
                "error": str(e),
                "logs": logs,
                "parsed_list": []
            }

import cv2
import numpy as np
import pytesseract
from pdf2image import convert_from_path
from datetime import datetime
import yaml

with open("config.yaml", "r", encoding="utf-8") as f:
    config = yaml.safe_load(f)

class OCRModule:
    def preprocess(self, img):
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        gray = cv2.convertScaleAbs(gray, alpha=config['ocr']['preprocessing']['alpha'], beta=0)
        thresh = cv2.adaptiveThreshold(
            gray, 255,
            cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
            cv2.THRESH_BINARY,
            config['ocr']['preprocessing']['adaptive_block_size'],
            config['ocr']['preprocessing']['adaptive_constant']
        )
        return thresh

    def extract(self, file_path):
        all_pages = []
        lang = config['ocr']['lang']

        if file_path.lower().endswith(".pdf"):
            images = convert_from_path(file_path)
            for page_num, img in enumerate(images):
                img_np = np.array(img)
                processed = self.preprocess(img_np)
                data = pytesseract.image_to_data(processed, lang=lang, output_type=pytesseract.Output.DICT)
                text = pytesseract.image_to_string(processed, lang=lang)
                blocks = []
                for i in range(len(data['text'])):
                    txt = data['text'][i].strip()
                    if txt:
                        conf = int(data['conf'][i]) if data['conf'][i] != -1 else 0
                        blocks.append({
                            "text": txt,
                            "left": int(data['left'][i]),
                            "top": int(data['top'][i]),
                            "width": int(data['width'][i]),
                            "height": int(data['height'][i]),
                            "confidence": conf
                        })
                all_pages.append({
                    "page": page_num + 1,
                    "text": text,
                    "blocks": blocks
                })
        else:
            img = cv2.imread(file_path)
            if img is None:
                raise ValueError(f"Could not read image file: {file_path}")
            processed = self.preprocess(img)
            data = pytesseract.image_to_data(processed, lang=lang, output_type=pytesseract.Output.DICT)
            text = pytesseract.image_to_string(processed, lang=lang)
            blocks = []
            for i in range(len(data['text'])):
                txt = data['text'][i].strip()
                if txt:
                    conf = int(data['conf'][i]) if data['conf'][i] != -1 else 0
                    blocks.append({
                        "text": txt,
                        "left": int(data['left'][i]),
                        "top": int(data['top'][i]),
                        "width": int(data['width'][i]),
                        "height": int(data['height'][i]),
                        "confidence": conf
                    })
            all_pages.append({
                "page": 1,
                "text": text,
                "blocks": blocks
            })

        return {
            "pages": all_pages,
            "timestamp": datetime.now().isoformat(),
            "ocr_engine": f"Tesseract {pytesseract.get_tesseract_version()}"
        }

import sqlite3
import json
from datetime import datetime

class DatabaseModule:
    def __init__(self, db_file="documents.db"):
        self.conn = sqlite3.connect(db_file, check_same_thread=False)
        self.create_tables()

    def create_tables(self):
        cur = self.conn.cursor()
        cur.execute("""
            CREATE TABLE IF NOT EXISTS documents (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                filename TEXT,
                ocr_text TEXT,
                data TEXT,
                created_at TEXT,
                ocr_version TEXT,
                llm_version TEXT
            )
        """)
        cur.execute("""
            CREATE TABLE IF NOT EXISTS logs (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                doc_id INTEGER,
                event TEXT,
                details TEXT,
                timestamp TEXT,
                FOREIGN KEY(doc_id) REFERENCES documents(id) ON DELETE SET NULL
            )
        """)
        self.conn.commit()

    def save(self, filename, ocr_text, parsed_data, ocr_version, llm_version):
        cur = self.conn.cursor()
        cur.execute("""
            INSERT INTO documents (filename, ocr_text, data, created_at, ocr_version, llm_version)
            VALUES (?, ?, ?, ?, ?, ?)
        """, (filename, ocr_text, json.dumps(parsed_data, ensure_ascii=False, indent=2),
              datetime.now().isoformat(), ocr_version, llm_version))
        self.conn.commit()
        return cur.lastrowid

    def log_event(self, doc_id, event, details=""):
        cur = self.conn.cursor()
        cur.execute("""
            INSERT INTO logs (doc_id, event, details, timestamp)
            VALUES (?, ?, ?, ?)
        """, (doc_id, event, details, datetime.now().isoformat()))
        self.conn.commit()

    def get_all(self):
        cur = self.conn.cursor()
        cur.execute("SELECT id, filename, created_at FROM documents ORDER BY id DESC")
        return cur.fetchall()

    def get_one(self, doc_id):
        cur = self.conn.cursor()
        cur.execute("SELECT ocr_text, data FROM documents WHERE id=?", (doc_id,))
        row = cur.fetchone()
        if row:
            return row[0], json.loads(row[1])
        return None

    def delete(self, doc_id):
        cur = self.conn.cursor()
        cur.execute("DELETE FROM documents WHERE id=?", (doc_id,))
        cur.execute("DELETE FROM logs WHERE doc_id=?", (doc_id,))
        self.conn.commit()

    def get_logs(self, doc_id):
        cur = self.conn.cursor()
        cur.execute("SELECT event, details, timestamp FROM logs WHERE doc_id=? ORDER BY timestamp", (doc_id,))
        return cur.fetchall()

    def update_document(self, doc_id, updated_data):
        cur = self.conn.cursor()
        cur.execute("SELECT data FROM documents WHERE id=?", (doc_id,))
        row = cur.fetchone()
        if not row:
            raise ValueError(f"Document {doc_id} not found")
        current_data = json.loads(row[0])
        current_data.update(updated_data)
        cur.execute("UPDATE documents SET data=? WHERE id=?",
                    (json.dumps(current_data, ensure_ascii=False, indent=2), doc_id))
        self.conn.commit()
